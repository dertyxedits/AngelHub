-- Helper: Create ESP Box
local function CreateESP(player)
 local char = player.Character
 if not char then return end

 local box = Instance.new("Model")
 box.Name = "ESP_".. player.Name
 box.Parent = workspace.CurrentCamera

 local billboardGui = Instance.new("BillboardGui")
 billboardGui.Name = "ESPBox"
 billboardGui.Size = UDim2.new(0, 100, 0, 100)
 billboardGui.StudsOffset = Vector3.new(0, 2, 0)
 billboardGui.AlwaysOnTop = true
 billboardGui.Parent = box

 local frame = Instance.new("Frame")
 frame.Name = "Frame"
 frame.Size = UDim2.new(0.8, 0, 0.8, 0)
 frame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
 frame.BackgroundTransparency = 0.5
 frame.Parent = billboardGui

 local corner = Instance.new("UICorner")
 corner.CornerRadius = UDim.new(0.5, 0)
 corner.Parent = frame

 local textLabel = Instance.new("TextLabel")
 textLabel.Size = UDim2.new(1, 0, 1, 0)
 textLabel.BackgroundTransparency = 1
 textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
 textLabel.TextStrokeTransparency = 0.5
 textLabel.Font = Enum.Font.GothamBold
 textLabel.TextSize = 14
 textLabel.Text = player.Name
 textLabel.TextScaled = true
 textLabel.Parent = billboardGui

 table.insert(ESPObjects, {Model = box, Player = player})
end

local function RemoveESP(player)
 for i, esp in pairs(ESPObjects) do
 if esp.Player == player then
 esp.Model:Destroy()
 table.remove(ESPObjects, i)
 break
 end
 end
end

-- Update ESP Loop
RunService.RenderStepped:Connect(function()
 if not Config.Visuals.ESP then return end

 for _, esp in pairs(ESPObjects) do
 local char = esp.Player.Character
 local humanoid = char and char:FindFirstChild("Humanoid")
 if not char or not humanoid or humanoid.Health <= 0 then
 RemoveESP(esp.Player)
 continue
 end

 local head = char:FindFirstChild("Head")
 local torso = char:FindFirstChild("Torso")

 if head and torso and billboardGui then
 local headPos = head.Position
 local torsoPos = torso.Position
 local distance = (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and (headPos - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude) or 999

 -- Position BillboardGui at head
 billboardGui.StudsOffset = Vector3.new(0, 2, 0)
 billboardGui.Size = UDim2.new(0, 100 + distance, 0, 100 + distance)

 -- Update Health Text
 if textLabel then
 textLabel.Text = string.format("%s\n%d HP\n%d studs", esp.Player.Name, math.floor(humanoid.Health), math.floor(distance))
 end
 end
 end
end)

-- Aimbot Logic
local function GetClosestTarget()
 local closest = nil
 local lowestDistance = math.huge

 for _, player in pairs(Players:GetPlayers()) do
 if player == LocalPlayer then continue end
 if player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
 if Config.Visuals.TeamCheck and player.Team == LocalPlayer.Team then continue end

 local targetPart = player.Character:FindFirstChild(Config.Aimbot.TargetPart)
 if targetPart then
 local distance = (targetPart.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude
 if distance < Config.Aimbot.FOV and distance < lowestDistance then
 lowestDistance = distance
 closest = targetPart
 end
 end
 end
 end
 return closest
end

local function GetCameraPos(part)
 return workspace.CurrentCamera:WorldToViewportPoint(part.Position)
end

RunService.RenderStepped:Connect(function()
 if not Config.Aimbot.Enabled then return end

 local target = GetClosestTarget()
 if target then
 local cameraPos = GetCameraPos(LocalPlayer.Character.HumanoidRootPart)
 local targetPos = GetCameraPos(target)

 local x = targetPos.X - cameraPos.X
 local y = targetPos.Y - cameraPos.Y

 local screenPos = workspace.CurrentCamera:ViewportPointToScreenPoint(cameraPos.X + x, cameraPos.Y + y, cameraPos.Z)

 -- Simple Mouse Movement
 local m = game:GetService("Players").LocalPlayer:GetMouse()
 local currentX, currentY = m.X, m.Y
 m.X = screenPos.X
 m.Y = screenPos.Y
 end
end)
