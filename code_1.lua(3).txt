local function slider(parent, labelText, x, y, key, minVal, maxVal, default)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(0, 300, 0, 52)
    container.Position = UDim2.new(0, x, 0, y)
    container.BackgroundTransparency = 1

    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, 0, 0, 18)
    label.BackgroundTransparency = 1
    label.Text = labelText .. ": " .. default
    label.TextColor3 = Color3.fromRGB(210, 210, 220)
    label.Font = Enum.Font.Gotham
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left

    local bar = Instance.new("Frame", container)
    bar.Size = UDim2.new(1, 0, 0, 6)
    bar.Position = UDim2.new(0, 0, 0, 26)
    bar.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    bar.BorderSizePixel = 0
    Instance.new("UICorner", bar).CornerRadius = UDim.new(0, 3)

    local fill = Instance.new("Frame", bar)
    fill.Size = UDim2.new((default - minVal) / (maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    fill.BorderSizePixel = 0
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 3)

    local knob = Instance.new("Frame", bar)
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new((default - minVal) / (maxVal - minVal), -7, 0.5, -7)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    Instance.new("UICorner", knob).CornerRadius = UDim.new(0, 4)

    local dragging = false
    local function set(input)
        local rel = input.Position.X - bar.AbsolutePosition.X
        rel = math.clamp(rel, 0, bar.AbsoluteSize.X)
        local pct = rel / math.max(bar.AbsoluteSize.X, 1)
        local val = math.floor(minVal + pct * (maxVal - minVal))
        fill.Size = UDim2.new(pct, 0, 1, 0)
        knob.Position = UDim2.new(pct, -7, 0.5, -7)
        cfg[key] = val
        label.Text = labelText .. ": " .. val
    end

    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            set(input)
        end
    end)
    bar.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            set(input)
        end
    end)
    bar.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    cfg[key] = default
end

-- Aimbot
local function getPart(char, partName)
    if not char then return nil end
    if partName == "Head" then return char:FindFirstChild("Head") end
    if partName == "HumanoidRootPart" then return char:FindFirstChild("HumanoidRootPart") end
    return char:FindFirstChild(partName) or char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
end

local function isAlive(player)
    local c = player.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function isEnemy(player)
    if not cfg.teamCheck then return true end
    local t1, t2 = lp.Team, player.Team
    if not t1 or not t2 then return true end
    return t1 ~= t2
end

local function getClosestTarget()
    local best = nil
    local bestDist = cfg.fov
    local center = camera.ViewportSize / 2

    for _, p in pairs(Players:GetPlayers()) do
        if p == lp then continue end
        if not isEnemy(p) then continue end
        if not isAlive(p) then continue end

        local char = p.Character
        local part = getPart(char, cfg.targetPart)
        if not part then continue end

        local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
        if not onScreen then continue end

        if cfg.visibilityCheck then
            local origin = camera.CFrame.Position
            local dir = (part.Position - origin).Unit
            local ray = Workspace:Raycast(origin, dir * (part.Position - origin).Magnitude)
            if ray and not ray.Instance:IsDescendantOf(char) then continue end
        end

        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
        if dist < bestDist then
            bestDist = dist
            best = part
        end
    end
    return best
end

-- ESP objects
local esp = {}

Players.PlayerRemoving:Connect(function(p)

        end

end)

-- Main loop
RunService.RenderStepped:Connect(function()
    -- Aimbot
    if cfg.aimbot then
        if not cfg.aimHold or UserInputService:IsKeyDown(cfg.aimKey) then
            local target = getClosestTarget()
            if target then
                local targetPos = target.Position
                local current = camera.CFrame.Position
                local look = CFrame.lookAt(current, targetPos)
                camera.CFrame = camera.CFrame:Lerp(look, cfg.smoothness / 10)
            end
        end
    end

    -- Fly
    if cfg.fly then
        local c = lp.Character
        if c then
            local hrp = c:FindFirstChild("HumanoidRootPart")
            local hum = c:FindFirstChildOfClass("Humanoid")
            if hrp and hum then
                hum.PlatformStand = true
                local move = hum.MoveDirection
                if move.Magnitude > 0 then move = move.Unit else move = Vector3.zero end
                local vel = move * cfg.flySpeed
                local vert = 0
                if cfg.flyUp then vert = cfg.flySpeed end
                if cfg.flyDown then vert = -cfg.flySpeed end
                if cfg.flyUp and cfg.flyDown then vert = 0 end
                hrp.AssemblyLinearVelocity = Vector3.new(vel.X, vert, vel.Z)
            end
        end
    end

    -- World
    if cfg.fullbright then
        Lighting.FogEnd = cfg.fogEnd
        Lighting.Brightness = cfg.brightness
        Lighting.ClockTime = cfg.clockTime
    end

    -- ESP
    for p, data in pairs(esp) do
        local char = p.Character
        local alive = char and char:FindFirstChildOfClass("Humanoid") and char.Humanoid.Health > 0
        local attach = char and (char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart"))

        -- Names
        if cfg.espNames and alive and attach then
            if not data.name then
                local bg = Instance.new("BillboardGui")
                bg.Size = UDim2.new(0, 120, 0, 20)
                bg.StudsOffset = Vector3.new(0, 2.5, 0)
                bg.AlwaysOnTop = true
                bg.Parent = gui
                local tl = Instance.new("TextLabel", bg)
                tl.Size = UDim2.new(1, 0, 1, 0)
                tl.BackgroundTransparency = 1
                tl.TextColor3 = Color3.fromRGB(255, 255, 255)
                tl.TextStrokeTransparency = 0.5
                tl.Font = Enum.Font.GothamBold
                tl.TextSize = 14
                data.name = bg
                data.nameLabel = tl
            end
            data.name.Adornee = attach
            data.nameLabel.Text = p.Name .. " [" .. math.floor(char.Humanoid.Health) .. "]"
            data.name.Enabled = true
        elseif data.name then
            data.name.Enabled = false
        end

        -- Box / Chams
        if (cfg.espBox or cfg.espChams) and alive and char then
            if not data.hl then
                local hl = Instance.new("Highlight")
                hl.Parent = char
                data.hl = hl
            end
            data.hl.FillColor = cfg.espChams and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 0, 0)
            data.hl.FillTransparency = cfg.espChams and 0.5 or 1
            data.hl.OutlineColor = cfg.espBox and Color3.fromRGB(0, 170, 255) or Color3.fromRGB(0, 0, 0)
            data.hl.OutlineTransparency = cfg.espBox and 0 or 1
            data.hl.Enabled = true
        elseif data.hl then
            data.hl.Enabled = false
        end

        -- Tracers
        if cfg.espTracers and alive and attach then
            if not data.tracer then
                local tr = Instance.new("Frame", gui)
                tr.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
                tr.BorderSizePixel = 0
                tr.AnchorPoint = Vector2.new(0.5, 0.5)
                data.tracer = tr
            end
            local pos, onScreen = camera:WorldToViewportPoint(attach.Position)
            if onScreen then
                local center = camera.ViewportSize / 2
                local delta = Vector2.new(pos.X, pos.Y) - center
                data.tracer.Size = UDim2.new(0, delta.Magnitude, 0, 2)
                data.tracer.Position = UDim2.new(0, center.X, 0, center.Y)
                data.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                data.tracer.Visible = true
            else
                data.tracer.Visible = false
            end
        elseif data.tracer then
            data.tracer.Visible = false
        end
    end

    -- Add new players to ESP table
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= lp and not esp[p] then esp[p] = {} end
    end
end)
