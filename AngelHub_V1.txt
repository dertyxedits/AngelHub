    local l = Instance.new("TextLabel")
    l.Position = UDim2.fromOffset(14, y)

local function toggle(text, y, key)
    local l = label(text, y, 15)
    local b = Instance.new("TextButton")
    b.Size = UDim2.fromOffset(58, 29)
    b.Position = UDim2.new(1, -72, 0, y)
    b.BackgroundColor3 = config[key] and Color3.fromRGB(45, 160, 190) or Color3.fromRGB(37, 48, 62)
    b.Text = config[key] and "ON" or "OFF"
    b.TextColor3 = Color3.fromRGB(235, 245, 255)
    b.Font = Enum.Font.GothamBold
    b.TextSize = 12
    b.Parent = panelScroll
    round(b, 14)
    b.MouseButton1Click:Connect(function()
        config[key] = not config[key]
        b.Text = config[key] and "ON" or "OFF"
        b.BackgroundColor3 = config[key] and Color3.fromRGB(45, 160, 190) or Color3.fromRGB(37, 48, 62)
    end)
    controls[key] = b
end

local function slider(text, y, key, minV, maxV, suffix)
    local l = label(text .. ": " .. tostring(config[key]) .. (suffix or ""), y)
    local bar = Instance.new("TextButton")
    bar.Position = UDim2.fromOffset(14, y + 29)
    bar.Size = UDim2.new(1, -28, 0, 12)
    bar.BackgroundColor3 = Color3.fromRGB(39, 53, 68)
    bar.Text = ""
    bar.AutoButtonColor = false
    bar.Parent = panelScroll
    round(bar, 6)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((config[key] - minV) / (maxV - minV), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(75, 220, 250)
    fill.BorderSizePixel = 0
    fill.Parent = bar
    round(fill, 6)

    local dragging = false
    local function setFromX(x)
        local p = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local value = math.floor(minV + p * (maxV - minV) + 0.5)
        config[key] = value
        fill.Size = UDim2.new(p, 0, 1, 0)
        l.Text = text .. ": " .. value .. (suffix or "")
    end
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

    round(b, 5)

    end)
end

local function buildTab(name)
    currentTab = name
    clearPanel()
    if name == "Aimbot" then
        section("AIM ASSIST SETTINGS")
        toggle("Enabled", 48, "enabled")
        slider("Aimbot Smoothness", 94, "smoothness", 0, 100, "%")
        slider("FOV Radius", 153, "fovRadius", 20, 500, " px")
        dropdown("Target Bone", 213, "targetPart", {"Head", "UpperTorso", "HumanoidRootPart"})
        toggle("Team Check", 258, "teamCheck")
        toggle("Draw FOV", 303, "drawFOV")
        label("Toggle lock: Q", 352, 13, Color3.fromRGB(115, 205, 225))
    elseif name == "Visuals" then
        section("PLAYER VISUALS")
        toggle("Player ESP labels", 48, "esp")
        label("Labels show name and distance.", 92, 13, Color3.fromRGB(145, 195, 215))
    elseif name == "World" then
        section("WORLD")
        label("World options can be added for your game.", 48, 13, Color3.fromRGB(145, 195, 215))
    elseif name == "Settings" then
        section("INTERFACE")
        label("RightShift: show / hide UI", 48, 13, Color3.fromRGB(145, 195, 215))
        label("Q: toggle lock-on", 78, 13, Color3.fromRGB(145, 195, 215))
    else
        section("MISC")
        label("AngelHub V1.0 • Studio game feature", 48, 13, Color3.fromRGB(145, 195, 215))
    end
    for tab, button in pairs(tabButtons) do
        button.BackgroundColor3 = tab == name and Color3.fromRGB(48, 115, 145) or Color3.fromRGB(28, 43, 60)
        button.TextColor3 = tab == name and Color3.fromRGB(220, 250, 255) or Color3.fromRGB(160, 198, 215)
    end
end

for name, button in pairs(tabButtons) do
    button.MouseButton1Click:Connect(function()
        buildTab(name)
    end)
end

local function sameTeam(other)
    return localPlayer.Team ~= nil and other.Team == localPlayer.Team
end

local function getPart(character)
    if not character then return nil end
    return character:FindFirstChild(config.targetPart) or character:FindFirstChild("HumanoidRootPart")
end

local espObjects = {}
local function removeESP(other)
    if espObjects[other] then
        espObjects[other]:Destroy()
        espObjects[other] = nil
    end
end

local function updateESP(other)
    if other == localPlayer or not config.esp then
        removeESP(other)
        return
    end
    local character = other.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local hum = character and character:FindFirstChildOfClass("Humanoid")
    if not root or not hum or hum.Health <= 0 then
        removeESP(other)
        return
    end
    if not espObjects[other] then
        local bb = Instance.new("BillboardGui")
        bb.Name = "AngelHubESP"
        bb.Adornee = root
        bb.Size = UDim2.fromOffset(180, 40)
        bb.StudsOffset = Vector3.new(0, 3, 0)
        bb.AlwaysOnTop = true
        local text = Instance.new("TextLabel")
        text.BackgroundTransparency = 1
        text.Size = UDim2.fromScale(1, 1)
        text.Font = Enum.Font.GothamBold
        text.TextSize = 13
        text.TextColor3 = Color3.fromRGB(100, 225, 255)
        text.TextStrokeTransparency = 0.25
        text.Parent = bb
        bb.Parent = gui
        espObjects[other] = bb
    end
    local bb = espObjects[other]
    bb.Adornee = root
    local labelObj = bb:FindFirstChildOfClass("TextLabel")
    local myRoot = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    local dist = myRoot and math.floor((root.Position - myRoot.Position).Magnitude) or 0
    if labelObj then labelObj.Text = other.DisplayName .. " [" .. dist .. " studs]" end
end

local function findTarget()
    local best, bestDist = nil, math.huge
    local center = camera.ViewportSize / 2
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= localPlayer and (not config.teamCheck or not sameTeam(other)) then
            local character = other.Character
            local hum = character and character:FindFirstChildOfClass("Humanoid")
            local part = getPart(character)
            if hum and hum.Health > 0 and part then
                local worldDist = (part.Position - camera.CFrame.Position).Magnitude
                local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
                local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if onScreen and worldDist <= config.maxDistance and screenDist <= config.fovRadius and screenDist < bestDist then
                    best, bestDist = other, screenDist
                end
            end
        end
    end
    return best
end

RunService.RenderStepped:Connect(function(dt)
    camera = workspace.CurrentCamera
    fov.Visible = config.drawFOV
    fov.Size = UDim2.fromOffset(config.fovRadius * 2, config.fovRadius * 2)

    for _, other in ipairs(Players:GetPlayers()) do
        updateESP(other)
    end
    for other in pairs(espObjects) do
        if not other.Parent then removeESP(other) end
    end

    if not config.enabled then
        selectedTarget = nil
        return
    end

    selectedTarget = findTarget()
    if selectedTarget then
        local part = getPart(selectedTarget.Character)
        if part then
            local aimPoint = part.Position
            local goal = CFrame.lookAt(camera.CFrame.Position, aimPoint)
            -- Smoothness 0 is responsive; 100 is slower.
            local responsiveness = 1 - (config.smoothness / 100)
            local alpha = math.clamp(responsiveness * dt * 18, 0.01, 1)
            camera.CFrame = camera.CFrame:Lerp(goal, alpha)
        end
    end
end)
